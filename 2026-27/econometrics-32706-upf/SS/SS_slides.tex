\begin{frame}{The problem}
  We want draws from a density known only up to a normalizing constant:
  \[
    \pi(x)=\frac{q(x)}{Z},\quad Z=\int_{\mathbb R}q(t)\,dt.
  \]
  We can evaluate \(q(x)\), but calculating \(Z\) or sampling
  directly from \(\pi\) may be difficult.
  \vspace{0.7em}

  \alert{Key idea:} represent \(\pi\) as the marginal distribution
  of a point chosen uniformly from a region under the graph of \(q\).
  This representation leads to slice sampling.
\end{frame}

\begin{frame}{Proposition: uniform sampling under a graph}
  Let \(q:\mathbb R\to[0,\infty)\) be measurable and
  \(0<Z=\int_{\mathbb R}q(x)\,dx<\infty\). Define
  \[
    A=\{(x,h)\in\mathbb R\times(0,\infty):0<h<q(x)\}.
  \]
  \begin{block}{Sampling proposition}
    If \((X,H)\) is uniform on \(A\) with respect to area,
    then \(X\) has density
    \[
      \pi(x)=\frac{q(x)}{Z}.
    \]
  \end{block}
  \vspace{0.5em}
  The height \(H\) is auxiliary: discard it after sampling
  \((X,H)\), and the remaining \(X\) follows the target distribution.
  \vspace{0.55em}

  {\tiny Source: Neal (2003), \emph{Slice sampling},
  \emph{Annals of Statistics} 31(3), 705--767,
  \href{https://doi.org/10.1214/aos/1056562461}{doi:10.1214/aos/1056562461}.}
\end{frame}

\begin{frame}{Proof of the sampling proposition}
  Adding vertical strips of height \(q(x)\) (Tonelli's theorem) gives
  \[
    |A|=\int_{\mathbb R}\int_0^{q(x)}dh\,dx
       =\int_{\mathbb R}q(x)\,dx=Z.
  \]
  Since \(|A|=Z\), the uniform density on \(A\) is
  \[
    p(x,h)=\frac{1}{|A|}\mathbf 1\{(x,h)\in A\}
          =\frac{1}{Z}\mathbf 1\{0<h<q(x)\}.
  \]
  Its marginal density for \(X\) is \(\pi\):
  \[
    p_X(x)=\int_0^{q(x)}\frac{1}{Z}\,dh
          =\frac{q(x)}{Z}=\pi(x).
  \]
  \alert{Conclusion 1:} sample \((X,H)\) from this joint density
  and keep only \(X\); then \(X\sim\pi\).
  \vspace{0.25em}

  \alert{Conclusion 2:} alternate draws from \(H\mid X\) and
  \(X\mid H\) to preserve this joint density.
\end{frame}

\begin{frame}{Conditional 1: the height given the position}
  From slide 4,
  \[
    p(x,h)=\frac{1}{Z}\mathbf 1\{0<h<q(x)\},\quad
    p_X(x)=\frac{q(x)}{Z}.
  \]
  For a fixed \(x\) with \(q(x)>0\), divide the joint density
  by the marginal density of \(X\):
  \[
    p(h\mid x)=\frac{p(x,h)}{p_X(x)}
      =\frac{1}{q(x)}\mathbf 1\{0<h<q(x)\}.
  \]
  So:
  \[
    H\mid X=x\sim \operatorname{Unif}(0,q(x)).
  \]
\end{frame}

\begin{frame}{Conditional 2: the position given the height}
  By definition, for a fixed height \(h>0\),
  \[
    x\in S_h\quad\Longleftrightarrow\quad q(x)>h
      \quad\Longleftrightarrow\quad 0<h<q(x)
      \quad\Longleftrightarrow\quad (x,h)\in A.
  \]
  Therefore the joint density can also be written as
  \[
    p(x,h)=\frac{1}{Z}\mathbf 1\{x\in S_h\}.
  \]
  For heights with \(|S_h|>0\), the height marginal is
  \(p_H(h)=|S_h|/Z\). Dividing gives
  \[
    p(x\mid h)=\frac{p(x,h)}{p_H(h)}
      =\frac{1}{|S_h|}\mathbf 1\{x\in S_h\}.
  \]
  So:
  \[
    X\mid H=h\sim \operatorname{Unif}(S_h).
  \]
\end{frame}

\begin{frame}{How to use it}
  The two conditional laws are
  \[
    H\mid X=x\sim \operatorname{Unif}(0,q(x))\quad\text{and}\quad
    X\mid H=h\sim \operatorname{Unif}(S_h).
  \]
  Alternate draws from these two conditionals. The resulting Markov
  chain has stationary joint density
  \[
    p(x,h)=\frac{1}{Z}\mathbf 1\{(x,h)\in A\}.
  \]
\end{frame}

\begin{frame}{The ideal slice-sampling update}
  Starting from the current state value \(x\):
  \begin{enumerate}
    \setlength{\itemsep}{0.65em}
    \item Draw \(U\sim \operatorname{Unif}(0,1)\) with realization \(u\),
          then set \(h=uq(x)\).
    \item Form the horizontal slice \(S_h=\{z:q(z)>h\}\).
    \item Draw \(X'\mid H=h\sim \operatorname{Unif}(S_h)\).
    \item Use the realized value \(x'\) as the next state and repeat.
  \end{enumerate}
\end{frame}

\begin{frame}{A simple numerical example}
  Let \(q(x)=e^{-x^2/2}\), the standard normal kernel, and
  suppose the current value is \(x=1\).
  For a realization \(u=\tfrac12\) of \(U\sim \operatorname{Unif}(0,1)\),
  the height is
  \[
    h=u\,q(1)=\tfrac12 e^{-1/2}.
  \]
  The slice condition is
  \[
    e^{-z^2/2}>\tfrac12e^{-1/2}
    \quad\Longleftrightarrow\quad
    |z|<\sqrt{1+2\log 2}\approx 1.54.
  \]
  Thus \(S_h\approx(-1.54,1.54)\), and \(X'\mid H=h\sim \operatorname{Unif}(S_h)\)
  has a realized value \(x'\) in this interval.
  \vspace{0.45em}

  The example illustrates one possible height; the next update
  draws a new one.
\end{frame}

\begin{frame}{Why the update targets \(\pi\)}
  Start with \(X\sim\pi\). Given its current value \(x\), draw
  \(H\mid X=x\); then \((X,H)\) has joint density \(p(x,h)\).
  Next draw \(X'\mid H=h\). This leaves the joint law unchanged,
  so discarding \(H\) gives \(X'\sim\pi\).
  \vspace{0.75em}

  \alert{This is a Gibbs-sampling argument.} It proves that \(\pi\)
  is stationary for the update; successive states need not be
  independent.
\end{frame}

\begin{frame}{When the slice is hard to sample from}
  The ideal update assumes we can draw uniformly from \(S_h\).
  For a complicated density, its endpoints may be unknown and
  \(S_h\) may have several pieces.
  \vspace{0.55em}

  A practical one-dimensional slice sampler constructs a bracket
  around the current point, tests candidate positions, and shrinks
  the bracket after rejections. The bracket rules must preserve the
  correct conditional distribution on the slice.
  \vspace{0.65em}

  The height step remains \(h=u\,q(x)\). If we work with log densities,
  the equivalent threshold is
  \[
    \log h=\log q(x)+\log u.
  \]
  This form avoids evaluating a tiny density value directly.
\end{frame}

\begin{frame}{What to remember}
  \begin{enumerate}
    \setlength{\itemsep}{0.8em}
    \item Uniform sampling under an unnormalized density curve
          gives the desired marginal distribution.
    \item A random height selects a horizontal slice; the position
          update moves within that slice.
    \item Alternating these conditional updates preserves the joint
          law and therefore the target marginal.
  \end{enumerate}
  \vspace{0.5em}

  {\small Reference: Radford M. Neal (2003),
  \emph{Slice sampling}, \emph{Annals of Statistics}
  31(3), 705--767,
  \href{https://doi.org/10.1214/aos/1056562461}{doi:10.1214/aos/1056562461}.}
\end{frame}


\begin{frame}{ESS uses the same slice idea}
  Let \(\phi_K(f)=\mathcal N(f;0,K)\). The target is
  \[
    \pi(f)\ \propto\ \phi_K(f)L(f).
  \]
  Introduce a height \emph{under the likelihood}:
  \[
    p(f,h)\ \propto\
      \phi_K(f)\,\mathbf 1\{0<h<L(f)\}.
  \]
  Ordinary slice sampling recovers its target by integrating out height:
  \[
    p_X(x)=\int_0^{q(x)}\frac{1}{Z}\,dh
      =\frac{q(x)}{Z}=\pi(x).
  \]
  ESS uses the same step, with the prior factor inside the integral:
  \[
    p_F(f)\propto\int_0^{L(f)}\phi_K(f)\,dh
      =\phi_K(f)L(f)\propto\pi(f).
  \]

  The two conditional distributions follow by dividing this joint
  density by its marginals.
\end{frame}

\begin{frame}{The two conditional updates for ESS}
  Let \(Z_* = \int \phi_K(z)L(z)\,dz\), the normalizing constant.
  The joint density is
  \[
    p(f,h)=\frac{1}{Z_*}\phi_K(f)\mathbf 1\{0<h<L(f)\}.
  \]
  The marginal density of \(F\) is \(p_F(f)=\phi_K(f)L(f)/Z_*\), so
  \[
    p(h\mid f)=\frac{p(f,h)}{p_F(f)}
      =\frac{1}{L(f)}\mathbf 1\{0<h<L(f)\},\quad
    H\mid F=f\sim \operatorname{Unif}(0,L(f)).
  \]
  At a fixed height, define
  \[
    C_h=\int \phi_K(z)\mathbf 1\{L(z)>h\}\,dz.
  \]
  Integrating over this slice gives \(p_H(h)=C_h/Z_*\).
  Dividing by this marginal gives
  \[
    p(f\mid h)=\frac{\phi_K(f)\mathbf 1\{L(f)>h\}}{C_h}
      =\pi_h(f),\quad F\mid H=h\sim\pi_h.
  \]
\end{frame}

\begin{frame}{ESS searches that slice along an ellipse}
  From the current value \(f\), draw \(V\sim\mathcal N(0,K)\)
  and \(U\sim \operatorname{Unif}(0,1)\).
  Let \(v\) and \(u\) be their realized values, set \(h=uL(f)\),
  and consider
  \[
    f(\theta)=f\cos\theta+v\sin\theta,\quad
    S_h^\theta=\{\theta:L(f(\theta))>h\}.
  \]
  A rotation of the \emph{pair} is
  \[
    (f_\theta,v_\theta)
      =(f\cos\theta+v\sin\theta,\,
        -f\sin\theta+v\cos\theta).
  \]
  ESS accepts only angles in \(S_h^\theta\). It shrinks a
  randomly placed angular bracket after rejected candidates;
  the next two slides show why this gives a valid update for
  \(p(f\mid h)\).
  \vspace{0.35em}

  Since \(f(0)=f\) and \(h<L(f)\), angle \(0\) is always eligible.
  The bracket keeps it inside while searching.
\end{frame}

\begin{frame}{The conditional distribution at fixed \(h\)}
  Let \(\phi_K\) be the \(\mathcal N(0,K)\) density and define
  \[
    C_h=\int \phi_K(z)\mathbf 1\{L(z)>h\}\,dz.
  \]
  Add an independent Gaussian draw \(V\) to the conditional target:
  \[
    \rho_h(f,v)
      =\frac{1}{C_h}\,
       \phi_K(f)\phi_K(v)\mathbf 1\{L(f)>h\}.
  \]
  Its marginal density for \(F\) is exactly the distribution we need:
  \[
    \int \rho_h(f,v)\,dv
      =\frac{\phi_K(f)\mathbf 1\{L(f)>h\}}{C_h}
      =p(f\mid h).
  \]
  Also, \(\rho_h(v\mid f)=\phi_K(v)\). Thus drawing
  \(V\sim\mathcal N(0,K)\) is an exact conditional draw for this
  augmented target.
\end{frame}

\begin{frame}{Why the ESS angle search preserves it}
  Rotating \((f,v)\) by \(\theta\) to \((f',v')\) has unit
  Jacobian and preserves the Gaussian product:
  \[
    \phi_K(f')\phi_K(v')=\phi_K(f)\phi_K(v).
  \]
  ESS accepts only \(L(f')>h\); both endpoints of an accepted
  move therefore have the same density under \(\rho_h\).
  \vspace{0.3em}

  The reverse rotation uses \(-\theta\). The randomly placed
  initial bracket and the shrinkage rule give matching
  probabilities to the forward and reverse angle paths
  (Murray, Adams and MacKay, 2010, Section 2.3).
  Hence the move satisfies detailed balance for \(\rho_h\).
  \vspace{0.35em}

  \alert{Consequently,}
  \[
    F\mid H=h\sim\pi_h
      \quad\Longrightarrow\quad
    F'_{\rm ESS}\mid H=h\sim\pi_h.
  \]
  This is a valid conditional \emph{Markov update}, not an
  independent draw from \(\pi_h\).
\end{frame}

\begin{frame}{Ordinary slice sampling versus ESS}
  \begin{center}
    \small
    \renewcommand{\arraystretch}{1.55}
    \begin{tabular}{@{}>{\raggedright\arraybackslash}p{1.7cm}
                    @{\hspace{0.3cm}}>{\raggedright\arraybackslash}p{4.25cm}
                    @{\hspace{0.3cm}}>{\raggedright\arraybackslash}p{4.25cm}@{}}
      \toprule
      & Ordinary slice sampling & Elliptical slice sampling\\
      \midrule
      Target
        & \(\pi(x)\propto q(x)\)
        & \(\pi(f)\propto\phi_K(f)L(f)\)\\
      Height
        & \(H\mid X=x\sim \operatorname{Unif}(0,q(x))\)
        & \(H\mid F=f\sim \operatorname{Unif}(0,L(f))\)\\
      Given \(h\)
        & Uniform on \(\{x:q(x)>h\}\)
        & Gaussian prior restricted to \(\{f:L(f)>h\}\)\\
      Update
        & Ideal: uniform conditional draw
        & Reversible Markov update on the restricted prior\\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{0.35em}

  \alert{Shared principle:} augment with a height, then update
  while preserving the conditional law on that slice.
  \vspace{0.35em}

       {proceedings.mlr.press/v9/murray10a.html}.}
\end{frame}
